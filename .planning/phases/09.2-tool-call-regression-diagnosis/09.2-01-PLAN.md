Structure it as a single Bash script that **both prints a table and exits non-zero on any failure**, so the Nyquist gate is automated.

    Script body (run as `.venv/bin/python - <<'PY' ... PY`):
    ```python
    import json, sys
    def flatten(path):
        d = json.load(open(path))
        return {(c["category"], b["benchmark"]): b["score"]
                for c in d["categories"] for b in c["benchmarks"]}
    old_custom = flatten("results/base_custom.json")
    new_custom = flatten("results/base_custom_repro.json")
    old_knowledge = flatten("results/base_knowledge.json")
    new_knowledge = flatten("results/base_knowledge_repro.json")
    old = {**old_custom, **old_knowledge}
    new = {**new_custom, **new_knowledge}
    rows = []
    any_fail = False
    for key in sorted(old):
        o = old[key]; n = new.get(key)
        if n is None:
            rows.append(f"{key!s:40s} old={o:.4f} new=MISSING FAIL")
            any_fail = True
            continue
        delta = abs(n - o)
        status = "PASS" if delta < 0.02 else "FAIL"
        if status == "FAIL":
            any_fail = True
        rows.append(f"{key!s:40s} old={o:.4f} new={n:.4f} delta={delta:.4f} {status}")
    print("=== D-07 Reproducibility Check ===")
    print("\n".join(rows))
    sys.exit(1 if any_fail else 0)
    ```
