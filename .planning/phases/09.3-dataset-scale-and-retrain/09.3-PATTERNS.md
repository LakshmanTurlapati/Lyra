#### Validation + write pattern (from `scripts/generate_knowledge_data.py` lines 1052-1074)

```python
def validate_batch(samples: list[dict]) -> dict:
    """Validate all samples in a batch via Conversation.model_validate()."""
    results = {"total": len(samples), "valid": 0, "invalid": 0, "errors": []}
    for i, sample in enumerate(samples):
        try:
            Conversation.model_validate(sample)
            results["valid"] += 1
        except Exception as e:
            results["invalid"] += 1
            results["errors"].append({"index": i, "error": str(e)})
    return results


def write_batch(samples: list[dict], output_path: Path) -> Path:
    """Write samples as JSONL to output_path. One JSON object per line."""
    output_path.parent.mkdir(parents=True, exist_ok=True)
    with open(output_path, "w", encoding="utf-8") as f:
        for sample in samples:
            f.write(json.dumps(sample) + "\n")
    return output_path
```

**Apply to gen_batch_09_3.py:** Reuse this exactly. 5. **Response parsing** — iterate `client.beta.messages.batches.results(batch_id)` (returns JSONL stream). Per-line: check `result.result.type == "succeeded"`, extract `msg.content[0].text`, JSON-decode that text into a ShareGPT `{"messages": [...], "tools": [...]}` dict, then validate.
